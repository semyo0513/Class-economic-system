function nav(panelId,el){
  document.querySelectorAll('.panel').forEach(p=>p.classList.remove('active'));
  document.getElementById('p-'+panelId).classList.add('active');
  if(el){document.querySelectorAll('.nav-item').forEach(i=>i.classList.remove('active'));el.classList.add('active');}
  if(panelId==='home')loadHome();
  else if(panelId==='lms'){
    // ★ 현재 활성된 서브탭만 로드 (급식 중복 표시 방지)
    const activeSubTab = document.querySelector('#p-lms .sub-tab.active');
    const activeTabText = activeSubTab ? activeSubTab.textContent.trim() : '';
    if(activeTabText.includes('호출')) loadCallList();
    else if(activeTabText.includes('숙제')) loadHwList();
    else if(activeTabText.includes('급식') || activeTabText.includes('시간표')) loadMealByDate();
    else if(activeTabText.includes('공지')) loadNotice();
    else if(activeTabText.includes('과제')) loadAssignments();
    else loadCallList(); // 기본값: 첫 번째 탭(호출)
  }
  else if(panelId==='eco')loadEcoStock();
  else if(panelId==='eval')loadEval();
  else if(panelId==='delegate')loadDelegatePanel();
}

async function loadMealByDate() {
  const dp = document.getElementById('meal-date-picker');
  if (!dp) return;
  const dateStr = dp.value || _dateToStr(new Date());
  const dateObj = new Date(dateStr);
  const korLabel = _formatKor(dateObj);

  // 날짜 레이블 업데이트
  const mealLabel = document.getElementById('meal-date-label');
  const ttLabel = document.getElementById('tt-date-label');
  if(mealLabel) mealLabel.textContent = korLabel;
  if(ttLabel) ttLabel.textContent = korLabel;

  // 급식 로드
  document.getElementById('meal-list').innerHTML = `<div style="text-align:center;padding:16px;color:var(--muted);font-size:12px">🔄 불러오는 중...</div>`;
  document.getElementById('tt-list').innerHTML = `<div style="text-align:center;padding:16px;color:var(--muted);font-size:12px">🔄 불러오는 중...</div>`;

  try {
    const meals = await api('getMeal', {date: dateStr});
    if (meals.success && meals.meals && meals.meals.length > 0) {
      const typeColors = {'조식':'var(--teal)','중식':'var(--gold)','석식':'var(--accent2)'};
      document.getElementById('meal-list').innerHTML = meals.meals.map(m => {
        const color = typeColors[m.type] || 'var(--accent)';
        const items = m.menu.split(/[,\n]/).map(s=>s.trim()).filter(Boolean);
        return `<div class="meal-section">
          <div class="meal-type-badge" style="background:${color}22;color:${color};border:1px solid ${color}44">${m.type}</div>
          <div>${items.map(item=>`<div class="menu-item">${item.replace(/\(\d+\)/g,'').trim()}</div>`).join('')}</div>
        </div>`;
      }).join('<div style="height:1px;background:var(--border);margin:8px 0"></div>');
    } else {
      document.getElementById('meal-list').innerHTML = `<div class="empty-state"><div class="empty-icon">🍽</div><p>${korLabel}<br>급식 정보가 없습니다</p></div>`;
    }
  } catch(e) {
    document.getElementById('meal-list').innerHTML = `<div class="empty-state"><div class="empty-icon">⚠️</div><p>급식 정보를 불러오지 못했습니다</p></div>`;
  }

  // 시간표 로드 (요일 기반)
  try {
    const tt = await api('getTimetable');
    const days = ['일','월','화','수','목','금','토'];
    const dayKorShort = days[dateObj.getDay()]; // "월","화"...
    const dayKorFull = dayKorShort + '요일';

    if (tt.success && tt.timetable && tt.timetable.length > 0) {
      const todayRows = tt.timetable.filter(t =>
        t.day === dayKorShort || t.day === dayKorFull
      );
      if (todayRows.length > 0) {
        const subjectColors = ['#6c8fff','#b48eff','#2ecc71','#f5c842','#e91e8c','#1abc9c','#f39c12','#e74c3c'];
        document.getElementById('tt-list').innerHTML = todayRows.map((t,i) =>
          `<div class="tt-row">
            <div class="tt-period" style="background:${subjectColors[i%subjectColors.length]}">${t.period}</div>
            <div><div class="tt-subject">${t.subject}</div>${t.teacher?`<div class="tt-teacher">${t.teacher}</div>`:''}
          </div></div>`
        ).join('');
      } else {
        document.getElementById('tt-list').innerHTML = `<div class="empty-state"><div class="empty-icon">📭</div><p>${dayKorFull} 시간표가 없습니다</p></div>`;
      }
    } else {
      document.getElementById('tt-list').innerHTML = `<div class="empty-state"><div class="empty-icon">📋</div><p>시간표 데이터가 없습니다<br><small style="font-size:10px">관리자 페이지에서 입력해 주세요</small></p></div>`;
    }
  } catch(e) {
    document.getElementById('tt-list').innerHTML = `<div class="empty-state"><div class="empty-icon">⚠️</div><p>시간표를 불러오지 못했습니다</p></div>`;
  }
}

  const canBuy = data.todayBought < data.dailyLimit && ME.cash >= G_LOTTERY_PRICE && data.ticketCount > 0;
  const buyBtn = document.getElementById('lottery-main-buy-btn');
  if (buyBtn) {
    buyBtn.disabled = !canBuy;

      if (data.ticketCount === 0)
        buyBtn.textContent = '⚠️ 복권 항목 없음 (관리자 등록 필요)';

      else
        buyBtn.textContent = '💸 잔액 부족';
    } else {
      buyBtn.textContent = '🎰 복권 1장 구매하기';
    }
  }

  // 미긁은 복권
  const scratches = data.unscratched || [];
  const scratchCard = document.getElementById('lottery-scratch-card');
  const scratchList = document.getElementById('lottery-unscratched-list');
  const cntEl = document.getElementById('lottery-scratch-count');
  if (scratches.length > 0) {

    scratchList.innerHTML = scratches.map(u => `

  }
}

function renderRelNetwork(data){
  const wrap=document.getElementById('rel-canvas-wrap');
  relW=wrap.clientWidth||340;relH=Math.min(relW*0.85,320);
  const canvas=document.getElementById('rel-canvas');canvas.width=relW;canvas.height=relH;
  const names=data.names||[];const N=names.length;if(!N)return;
  const cx=relW/2,cy=relH/2,r=Math.min(relW,relH)*0.36;
  relNodes=names.map((name,i)=>{const a=(2*Math.PI*i/N)-Math.PI/2;return{name,x:cx+r*Math.cos(a),y:cy+r*Math.sin(a),base:{x:cx+r*Math.cos(a),y:cy+r*Math.sin(a)}};});
  const combined=data.combined||[];

  selRelNode=null;

const relEdgeColor={tx:'rgba(108,143,255,',seat:'rgba(26,188,156,',emo:'rgba(233,30,140,',mixed:'rgba(180,142,255,'};
function animRelNet(canvas){
  const ctx=canvas.getContext('2d');let t=0;
  function draw(){
    t+=0.012;ctx.clearRect(0,0,relW,relH);
    ctx.strokeStyle='rgba(38,48,71,.35)';ctx.lineWidth=.5;
    for(let x=0;x<relW;x+=30){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,relH);ctx.stroke();}
    for(let y=0;y<relH;y+=30){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(relW,y);ctx.stroke();}
    const N=relNodes.length;
    relNodes.forEach((n,i)=>{const a=(2*Math.PI*i/N)-Math.PI/2+Math.sin(t+i*.9)*.03;const rr=Math.min(relW,relH)*.36+Math.sin(t*1.1+i)*1.5;n.x=relW/2+rr*Math.cos(a);n.y=relH/2+rr*Math.sin(a);});
    relEdges.forEach(e=>{
      const fn=relNodes[e.fi],tn=relNodes[e.ti];if(!fn||!tn)return;
      const isHL=selRelNode!==null&&(e.fi===selRelNode||e.ti===selRelNode);const fade=selRelNode!==null&&!isHL;
      const alpha=fade?.04:isHL?.9:.3;const lw=Math.min(e.weight/18+1,5)*(isHL?1.8:1);
      const c=relEdgeColor[e.type]||relEdgeColor.mixed;
      const midX=(fn.x+tn.x)/2+Math.sin(t*.8+e.fi)*4,midY=(fn.y+tn.y)/2+Math.cos(t*.8+e.ti)*4;
      ctx.beginPath();ctx.moveTo(fn.x,fn.y);ctx.quadraticCurveTo(midX,midY,tn.x,tn.y);
      ctx.strokeStyle=c+alpha+')';ctx.lineWidth=lw;ctx.stroke();
      if(isHL){ctx.fillStyle='rgba(255,255,255,.75)';ctx.font='bold 8px Noto Sans KR';ctx.textAlign='center';ctx.fillText(e.weight,midX,midY-3);}
    });
    relNodes.forEach((n,i)=>{
      const isSel=i===selRelNode;const isConn=selRelNode!==null&&relEdges.some(e=>(e.fi===selRelNode&&e.ti===i)||(e.ti===selRelNode&&e.fi===i));const fade=selRelNode!==null&&!isSel&&!isConn;
      if(isSel){const w=(t*2)%1;ctx.beginPath();ctx.arc(n.x,n.y,14+w*12,0,Math.PI*2);ctx.strokeStyle=`rgba(180,142,255,${.5*(1-w)})`;ctx.lineWidth=2;ctx.stroke();}
      ctx.globalAlpha=fade?.18:1;ctx.beginPath();ctx.arc(n.x,n.y,12,0,Math.PI*2);
      const g=ctx.createRadialGradient(n.x-2,n.y-2,1,n.x,n.y,12);
      if(isSel){g.addColorStop(0,'#e91e8c');g.addColorStop(1,'#880e4f');}else if(isConn){g.addColorStop(0,'#b48eff');g.addColorStop(1,'#6a0dad');}else{g.addColorStop(0,'#2a3550');g.addColorStop(1,'#161c28');}
      ctx.fillStyle=g;ctx.fill();ctx.strokeStyle=isSel?'rgba(233,30,140,.9)':isConn?'rgba(180,142,255,.8)':'rgba(108,143,255,.4)';ctx.lineWidth=isSel?2:1.5;ctx.stroke();
      ctx.fillStyle=isSel?'#ff80b5':isConn?'#d4a8ff':'rgba(221,228,240,.92)';ctx.font=(isSel?'bold ':'')+'9px Noto Sans KR';ctx.textAlign='center';ctx.textBaseline='middle';
      ctx.fillText(n.name.length>3?n.name.slice(0,3):n.name,n.x,n.y);ctx.globalAlpha=1;
    });
    relAnimId=requestAnimationFrame(draw);
  }draw();
}

function doAdmGrantPerm(btn){
  const name = document.getElementById('adm-perm-sel').value;
  if(!name) return showToast('학생을 선택해주세요','var(--red)');
  const permChecks = ['perm-salary','perm-fine','perm-tax','perm-notice','perm-praise','perm-warning', 'perm-mart'];
const permValues = ['월급배부','벌금징수','세금징수','공지작성','칭찬카드','경고', '마트관리'];
  const selected = [];
  permChecks.forEach((id,i)=>{if(document.getElementById(id)&&document.getElementById(id).checked)selected.push(permValues[i]);});
  const permissions = selected.join(',');
  wrapClick(btn,async()=>{
    const res=await api('adminGrantPermission',{studentName:name,permissions});
    showToast(res.msg,res.success?'var(--green)':'var(--red)');
    if(res.success){
      // 체크박스 초기화
      permChecks.forEach(id=>{const el=document.getElementById(id);if(el)el.checked=false;});
      loadAdminData();
    }
  });
}

    showToast(res.msg,res.success?'var(--green)':'var(--red)');
    if(res.success)loadAdminData();
  });
}

function admSeatClick(id,isActive){
  const now=Date.now();const last=ADM_SEAT_CLICK_TIMER[id]||0;const diff=now-last;ADM_SEAT_CLICK_TIMER[id]=now;
  if(diff<380){
    if(String(isActive).trim()!=='O')return;
    const name=prompt('배정할 학생 이름:');if(!name)return;
    api('adminSetSeat',{id,name}).then(res=>{if(res.success){showToast(`${name} 배정`,'var(--green)');loadAdminData();}});
  }else{
    setTimeout(()=>{if(Date.now()-ADM_SEAT_CLICK_TIMER[id]<380)return;if(!confirm(String(isActive).trim()==='O'?'이 자리를 비활성(복도)로 전환?':'이 자리를 활성화?'))return;api('adminToggleSeat',{id}).then(res=>{if(res.success){showToast('전환됨','var(--green)');loadAdminData();}});},400);}
}

async function loadAdmLottery(btn){
  if(btn)btn.disabled=true;
  const data=await api('adminGetLotteryData');
  if(btn)btn.disabled=false;
  const list=document.getElementById('adm-lottery-list');
  const tickets=data.tickets||[];
  if(!tickets.length){list.innerHTML='<div style="color:var(--muted);padding:10px">등록된 복권 없음</div>';return;}
  const totalProb=tickets.reduce((a,t)=>a+Number(t['확률']||0),0);
  list.innerHTML=tickets.map(t=>{
    const prob=Number(t['확률']||0);
    const pct=totalProb>0?(prob/totalProb*100).toFixed(1):'0.0';
    const cash=Number(t['현금']||0);
    return `<div class="card2" style="margin-bottom:6px">
      <div style="display:flex;justify-content:space-between;align-items:flex-start">
        <div>
          <span class="badge" style="background:rgba(108,143,255,.2);color:var(--accent);margin-right:4px">${t['구분']}</span>
          <b>${t['제목']}</b>
          ${cash>0?`<span style="color:var(--gold);font-size:10px;margin-left:4px">💰${cash.toLocaleString()}원</span>`:''}
          ${t['내용']?`<span style="color:var(--muted);font-size:10px;margin-left:4px">(${t['내용']})</span>`:''}
        </div>
        <button class="btn btn-red btn-xs" onclick="doAdmDelLottery('${t['제목']}','${t['구분']}')">삭제</button>
      </div>
      <div style="margin-top:5px;font-size:10px;color:var(--muted)">
        확률: ${prob} (${pct}%)
        <div style="background:var(--border);border-radius:3px;height:4px;margin-top:3px;overflow:hidden">
          <div style="background:var(--gold);height:100%;width:${pct}%;border-radius:3px"></div>
        </div>
      </div>
    </div>`;
  }).join('');
  // 총 확률 표시
  list.insertAdjacentHTML('afterbegin',`<div style="font-size:11px;color:var(--muted);margin-bottom:10px;padding:8px;background:var(--card2);border-radius:8px">
    📊 총 확률 합계: <b style="color:var(--gold)">${totalProb}</b> (각 항목의 당첨 비율로 사용됩니다)
  </div>`);
}

// 파일 선택/드래그 처리 (Drive 업로드 버전)
function asgHandleFile(file){
  if(!file) return;
  const ext = file.name.split('.').pop();
  const today = new Date();
  const dateStr = today.toLocaleDateString('sv-SE').replace(/-/g,'');
  const zone = document.getElementById('asg-file-drop-zone');
  const filetype = zone.dataset.filetype || '자유';
  const autoName = `${ME.name}_${filetype}_${dateStr}_1교시.${ext}`;
  document.getElementById('asg-selected-filename').textContent = `📎 ${autoName}`;
  zone.dataset.autoname = autoName;
  zone.dataset.file = file.name; // 원본 파일명 저장
  window._asgSelectedFile = file; // ★ 실제 파일 객체 저장
  document.getElementById('asg-selected-filename').style.display = 'block';
  document.getElementById('asg-submit-confirm-btn').disabled = false;
}

// 실제 제출 처리 (Drive 업로드 버전)
async function doSubmitConfirm(btn){
  const zone = document.getElementById('asg-file-drop-zone');
  const assignmentId = zone.dataset.assignmentId;
  const autoName = zone.dataset.autoname || '';
  const file = window._asgSelectedFile;

  if(!file) return showToast('파일을 먼저 선택하세요', 'var(--red)');

  btn.disabled = true;
  showLoad('파일 업로드 중...');
  try {
    // 1. FileReader로 base64 변환
    const base64Data = await new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = e => resolve(e.target.result.split(',')[1]);
      reader.onerror = () => reject(new Error('파일 읽기 실패'));
      reader.readAsDataURL(file);
    });

    // 2. Google Drive 업로드
    const upRes = await api('uploadAssignmentFile', {
      name: ME.name,
      assignmentId,
      base64Data,
      fileName: autoName || file.name,
      mimeType: file.type || 'application/octet-stream'
    });

    if(!upRes || !upRes.success){
      hideLoad();
      showToast(upRes?.msg || '업로드 실패', 'var(--red)');
      btn.disabled = false;
      return;
    }

    // 3. 제출 기록 (Drive URL 저장)
    const res = await api('submitAssignment', {
      name: ME.name,
      assignmentId,
      memo: upRes.fileUrl
    });

    hideLoad();
    showToast(res.msg, res.success ? 'var(--green)' : 'var(--red)');
    if(res.success){
      closePopup('asg-submit-popup');
      window._asgSelectedFile = null;
      loadHome();
      loadAssignments();
    }
  } catch(e) {
    hideLoad();
    showToast('오류: ' + e.message, 'var(--red)');
  } finally {
    btn.disabled = false;
  }
}

        <span style="color:var(--green);font-weight:700">+${fmt(t['금액'])}원</span>
      </div>
    `).join('')
    : '<div style="color:var(--muted);text-align:center;padding:10px">아직 결제 내역이 없습니다.</div>';
}

function doAddMartItem(btn) {
  wrapClick(btn, async () => {
    const name = document.getElementById('dlg-mart-item').value;
    const price = document.getElementById('dlg-mart-price').value;
    const stock = document.getElementById('dlg-mart-stock').value;
    
    if (!name || !price || !stock) return showToast('모든 항목(이름, 가격, 수량)을 입력하세요', 'var(--red)');
    
    const res = await api('addMartItem', { name: ME.name, itemName: name, price: price, stock: stock });
    showToast(res.msg, res.success ? 'var(--green)' : 'var(--red)');
    
    if (res.success) {
      document.getElementById('dlg-mart-item').value = '';
      document.getElementById('dlg-mart-price').value = '';
      document.getElementById('dlg-mart-stock').value = '';
      loadMartStats(); // 통계 새로고침
    }
  });
}


// 관리자용 마트 물품 목록 (삭제/재고수정)
async function loadAdmMartItems() {
  const items = await api('getMartItems');
  const el = document.getElementById('adm-mart-item-list');
  if (!items || items.length === 0) {
    el.innerHTML = '<div style="color:var(--muted);padding:8px">등록된 물품이 없습니다.</div>';
    return;
  }
  el.innerHTML = items.map(i => `
    <div class="card2" style="display:flex;justify-content:space-between;align-items:center;margin-bottom:6px;gap:8px">
      <div style="flex:1">
        <div style="font-weight:700">${i.아이템명}</div>
        <div style="font-size:10px;color:var(--muted)">${fmt(i.가격)}원 | 재고: ${i.재고}개</div>
      </div>
      <div style="display:flex;gap:4px;align-items:center">
        <input type="number" id="adm-stock-input-${i.아이템명}" value="${i.재고}" min="0"
               style="width:56px;padding:4px 6px;font-size:11px;margin:0">
        <button class="btn btn-teal btn-xs" onclick="doAdmUpdateMartStock('${i.아이템명}', this)">수정</button>
        <button class="btn btn-red btn-xs" onclick="doAdmDeleteMartItem('${i.아이템명}', this)">삭제</button>
      </div>
    </div>
  `).join('');
}

// 관리자 - 마트 재고 수정
function doAdmUpdateMartStock(itemName, btn) {
  wrapClick(btn, async () => {
    const newStock = document.getElementById(`adm-stock-input-${itemName}`).value;
    if (newStock === '' || Number(newStock) < 0) return showToast('수량을 올바르게 입력하세요', 'var(--red)');
    const res = await api('updateMartStock', { itemName: itemName, newStock: newStock });
    showToast(res.msg, res.success ? 'var(--green)' : 'var(--red)');
    if (res.success) loadAdmMartItems();
  });
}

// 관리자 - 마트 물품 삭제
function doAdmDeleteMartItem(itemName, btn) {
  wrapClick(btn, async () => {
    if (!confirm(`"${itemName}" 물품을 삭제하시겠습니까?`)) return;
    const res = await api('deleteMartItem', { itemName: itemName });
    showToast(res.msg, res.success ? 'var(--green)' : 'var(--red)');
    if (res.success) loadAdmMartItems();
  });
}
